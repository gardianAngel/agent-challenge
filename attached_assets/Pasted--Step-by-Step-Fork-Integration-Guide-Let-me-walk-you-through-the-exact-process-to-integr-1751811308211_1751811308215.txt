### **Step 5: Adapt Your Multi-Agent System**

Create the main DeFi Risk Oracle agent:

```typescript
// src/agents/defi-risk-oracle/index.ts
import { Agent } from '@mastra/core';
import { z } from 'zod';

export const defiRiskOracleAgent = new Agent({
  name: 'DeFi Risk Oracle',
  instructions: `
    You are a sophisticated DeFi risk assessment system that monitors 
    blockchain protocols, smart contracts, and market conditions to 
    provide real-time risk scores and alerts.
  `,
  model: {
    provider: 'openai',
    name: 'gpt-4',
  },
  tools: {
    // Your existing tools adapted for Mastra
    analyzeProtocol: {
      description: 'Analyze a DeFi protocol for various risk factors',
      parameters: z.object({
        protocolAddress: z.string(),
        chain: z.enum(['ethereum', 'polygon', 'arbitrum', 'solana']),
      }),
      execute: async ({ protocolAddress, chain }) => {
        // Your existing protocol analysis logic
        return await analyzeProtocolRisk(protocolAddress, chain);
      },
    },
    
    monitorSmartContract: {
      description: 'Monitor smart contract for vulnerabilities',
      parameters: z.object({
        contractAddress: z.string(),
        chain: z.string(),
      }),
      execute: async ({ contractAddress, chain }) => {
        // Your existing smart contract analysis
        return await analyzeSmartContract(contractAddress, chain);
      },
    },
    
    assessMarketRisk: {
      description: 'Assess market risk conditions',
      parameters: z.object({
        protocols: z.array(z.string()),
      }),
      execute: async ({ protocols }) => {
        // Your existing market risk analysis
        return await assessMarketConditions(protocols);
      },
    },
    
    generateAlerts: {
      description: 'Generate risk alerts based on thresholds',
      parameters: z.object({
        riskScores: z.record(z.number()),
        thresholds: z.record(z.number()),
      }),
      execute: async ({ riskScores, thresholds }) => {
        // Your existing alert generation
        return await generateRiskAlerts(riskScores, thresholds);
      },
    },
  },
});
```

### **Step 9: Create Docker Configuration**

```dockerfile
# Dockerfile (in root directory)
FROM node:18-alpine AS builder

WORKDIR /app
COPY package*.json ./
COPY pnpm-lock.yaml ./

RUN npm install -g pnpm
RUN pnpm install

COPY . .
RUN pnpm run build

FROM node:18-alpine AS runner

WORKDIR /app

# Copy built application
COPY --from=builder /app/dist ./dist
COPY --from=builder /app/package.json ./
COPY --from=builder /app/pnpm-lock.yaml ./

RUN npm install -g pnpm
RUN pnpm install --production

EXPOSE 3000

CMD ["node", "dist/index.js"]
```
